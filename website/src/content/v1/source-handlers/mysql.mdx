# MySQL

Use `@omnigraph/sql` with `type: 'mysql'`. One transport kind (`sql`) covers MySQL, SQLite, and
Postgres; dialect comes from loader `type` / the connection URI.

```sh npm2yarn
npm i @omnigraph/sql
```

```ts filename="mesh.config.ts"
import { defineConfig } from '@graphql-mesh/compose-cli'
import { loadSqlSubgraph } from '@omnigraph/sql'

export const composeConfig = defineConfig({
  subgraphs: [
    {
      sourceHandler: loadSqlSubgraph('Employees', {
        type: 'mysql',
        endpoint: 'mysql://root:passwd@localhost:3306/employees'
        // endpoint: `mysql://${process.env.MYSQL_USER}:${process.env.MYSQL_PASSWORD}@${process.env.MYSQL_HOST}:${process.env.MYSQL_PORT}/${process.env.MYSQL_DATABASE}`
      })
    }
  ]
})
```

<Callout type="info">
  This replaces `@omnigraph/mysql` / `@mysqlSelect` and legacy `@graphql-mesh/mysql`. Filters are
  typed (`StringFilter` / `IntFilter`), not opaque String where-bags.
</Callout>
